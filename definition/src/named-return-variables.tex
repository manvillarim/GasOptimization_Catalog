\begin{refactoring}[H]
\begin{footnotesize}
\caption{Use named return variables.}
\label{rule:named-returns}
\begin{tabular}{lll}
   \framebox[0.45\textwidth]{
      \begin{scriptsize}
      \begin{tabular}{l}
         $[\ldots]$\\
         \texttt{contract} $\langle A\rangle$ \texttt{\{}\\
         \hspace{0.5cm} $[\ldots]$\\
         \hspace{0.5cm} \texttt{function} $\langle f\rangle$\texttt{(}$\langle\mathit{parameters}\rangle$\texttt{)} $[\ldots]$\\
         \hspace{1cm} \texttt{returns (}$\langle T\rangle$\texttt{)} \texttt{\{}\\
         \hspace{1cm} $\langle T\rangle$ $\langle v\rangle$\texttt{;}\\
         \hspace{1cm} $\langle P\rangle$\\
         \hspace{1cm} \texttt{return} $\langle v\rangle$\texttt{;}\\
         \hspace{0.5cm} \texttt{\}}\\
         \hspace{0.5cm} $[\ldots]$\\
         \texttt{\}}\\
         $[\ldots]$\\
      \end{tabular}
      \end{scriptsize}
   }

   & $=$&

\framebox[0.45\textwidth]{
   \begin{scriptsize}
   \begin{tabular}{l}
      $[\ldots]$\\
      \texttt{contract} $\langle A\rangle$ \texttt{\{}\\
      \hspace{0.5cm} $[\ldots]$\\
      \hspace{0.5cm} \texttt{function} $\langle f\rangle$\texttt{(}$\langle\mathit{parameters}\rangle$\texttt{)} $[\ldots]$\\
      \hspace{1cm} \texttt{returns (}$\langle T\rangle$ $\langle v\rangle$\texttt{)} \texttt{\{}\\
      \hspace{1cm} $\langle P\rangle$\\
      \hspace{0.5cm} \texttt{\}}\\
      \hspace{0.5cm} $[\ldots]$\\
      \texttt{\}}\\
      $[\ldots]$\\
      \\[2ex]
   \end{tabular}
   \end{scriptsize}
}
\end{tabular}
\justifying

\mbox{}\\

\noindent \textbf{where} $A$, $f$ and $v$ are identifiers, $T$ is a type, $P$ is a Solidity command, $\mathit{parameters}$ is a list of identifiers, and $[\ldots]$ denotes Solidity constructs;

\noindent \textbf{provided} $v$ does not occur in $\langle\mathit{parameters}\rangle$ nor as a declaration inside $P$; and every \texttt{return} statement occurring in $P$ carries an explicit expression.
\end{footnotesize}
\end{refactoring}
